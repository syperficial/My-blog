2. 🎯 本地验证码逻辑实现 ​

  • 通过 setInterval 实现倒计时 ⏳
  • 点击时随机生成4位验证码，防止重复点击 🔄
  • 页面卸载时清除定时器，避免内存泄漏 🧹
js
export const useCaptcha = (timeout) => {
    const countdown = ref(timeout)       // 倒计时初始值
    const generateCode = ref('')         // 随机验证码字符串
    let timer = null                     // 定时器标识

    // 生成4位随机验证码
    const getCode = () => {
        generateCode.value = Math.floor(1000 + Math.random() * 9000).toString()
    }

    // 启动倒计时定时器
    const startTimer = () => {
        clearTimer()
        countdown.value = timeout
        timer = setInterval(() => {
            countdown.value--
            if (countdown.value <= 0) {
                getCode()
                countdown.value = timeout
            }
        }, 1000)
    }

    // 计算是否可点击（防止重复点击）
    const isClickable = computed(() => countdown.value <= timeout)

    // 清除定时器
    const clearTimer = () => {
        if (timer) {
            clearInterval(timer)
            timer = null
        }
    }

    onMounted(() => {
        getCode()
        startTimer()
    })

    onUnmounted(() => clearTimer())

    return {
        countdown,
        generateCode,
        getCode,
        startTimer,
        isClickable,
        clearTimer
    }
}
