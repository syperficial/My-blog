权限校验封装函数 ​

js
const roleAliasMap = {
  super: '超级管理员',
  user: '普通用户',
  auth: '已认证',
}

export const hasPermission = (roles = []) => {
  const user = useUserStore().fullUserInfo || {}
  if (user.isSuperAdmin) return true

  const userRole = user.role || ''
  const userStatuses = user.statuses?.map(s => s.text) || []

  return roles.some(role => {
    const actual = roleAliasMap[role] || role
    return userRole === actual || userStatuses.includes(actual)
  })
}
