🆔 实名认证 ​

  • 判断认证状态，支持提交实名认证信息。
  • 校验身份证格式，更新状态标签。
js
const isVerified = computed(() => {
  return userStore.fullUserInfo.statuses?.some((s) => s.text === '已认证')
})

const submitRealNameForm = async () => {
  try {
    await realNameFormRef.value.validate()
    const statuses = [{ text: '已认证', type: 'success-tag' }]
    const payload = {
      statuses,
      realName: realNameForm.value.name,
      idNumber: realNameForm.value.idNumber,
    }
    await updateUserInfo(userStore.id, payload)
    userStore.loadUser({
      ...userStore.fullUserInfo,
      ...payload,
      id: userStore.id,
    })
    dialogVisible.value = false
    resetDialog()
    ElMessage.success('实名认证成功')
  } catch (error) {
    ElMessage.error('实名认证失败')
  }
}
